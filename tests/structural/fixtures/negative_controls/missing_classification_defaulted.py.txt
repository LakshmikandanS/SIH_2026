# NEGATIVE CONTROL for test_profile_ceiling_enforced_at_ingest.
# A document without classification and ACL metadata is rejected, loudly, never
# defaulted (packages/knowledge/AGENTS.md). This file exists so the detector can prove
# it still catches that edge.

from __future__ import annotations


def normalise_upload(raw: dict) -> dict:
    classification = raw.get("classification", "PUBLIC")   # forbidden -- silent default
    acl = raw.get("acl") or []                              # forbidden -- silent default
    return {"classification": classification, "acl": acl, "text": raw["text"]}
