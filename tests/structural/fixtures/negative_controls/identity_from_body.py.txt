# NEGATIVE CONTROL for test_identity_not_from_body.
# Identity is never read from a request body -- it comes from the verified token the
# identity middleware attaches, never from a field the caller supplied. This file
# exists so the detector can prove it still catches that edge.

from __future__ import annotations


async def create_task(request):
    body = await request.json()
    user_id = body["user_id"]          # forbidden -- identity read from the request body
    department = body.get("department")  # forbidden -- same mistake, defaulted
    return {"user_id": user_id, "department": department}
