# NEGATIVE CONTROL for test_no_egress.
# Only the gateway's declared inference clients make outbound network calls; nothing
# else does, and nothing calls an external host directly. This file exists so the
# detector can prove it still catches that edge.

from __future__ import annotations

import requests


def fetch_something():
    return requests.get("https://api.example.com/v1/status")   # forbidden -- raw egress


async def fetch_async():
    import httpx

    async with httpx.AsyncClient() as client:
        return await client.get("https://example.com/ping")    # forbidden -- raw egress
